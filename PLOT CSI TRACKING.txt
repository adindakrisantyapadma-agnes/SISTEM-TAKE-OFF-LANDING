def extract_csi(df, rx_name):
    df_rx = df[df["rx"] == rx_name]
    packets = []

    for _, row in df_rx["csi_raw"].items():
        if not isinstance(row, str):
            continue
        try:
            match = re.search(r'\[(.*?)\]', row)
            if match:
                values = list(map(float, match.group(1).split(",")))
                packets.append(values)
        except:
            continue

    if not packets:
        print(f"⚠️ No data for {rx_name}")
        return np.array([])

    min_len = min(len(p) for p in packets)
    packets = [p[:min_len] for p in packets]

    print(f"✅ {rx_name}: {len(packets)} packets")
    return np.array(packets)

# 2. PREPROCESS
# =========================
def normalize_csi(csi):
    if len(csi) == 0:
        return csi
    return (csi - np.mean(csi, axis=0)) / (np.std(csi, axis=0) + 1e-8)


def filter_noise(csi, window=5):
    if len(csi) == 0:
        return csi

    out = np.zeros_like(csi)
    for i in range(csi.shape[1]):
        out[:, i] = np.convolve(csi[:, i], np.ones(window) / window, mode='same')
    return out

csi_rx4 = filter_noise(normalize_csi(csi_rx4), 7)

# =========================
# 3. DETECTOR
# =========================
class CSIDetector:
    def __init__(self, csi, name):
        self.csi = csi
        self.name = name

    def motion(self):
        if len(self.csi) < 2:
            return np.array([])
        diff = np.abs(np.diff(self.csi, axis=0))
        m = np.median(diff, axis=1)
        if np.max(m) > 0:
            m = m / np.max(m) * 100
        return m

# =========================
# 4. FUSION
# =========================
class Fusion:
    def __init__(self, detectors):
        self.detectors = detectors

    def get_real_mask(self):
        motions = []

        for d in self.detectors:
            m = d.motion()
            if len(m) > 0:
                thr = np.mean(m) + np.std(m)
                motions.append(m > thr)

        min_len = min(len(m) for m in motions)
        motions = [m[:min_len] for m in motions]

        stacked = np.stack(motions)
        consensus = np.sum(stacked, axis=0)

        real = consensus >= 3
        return real, motions

# =========================
# 5. LOCALIZER (IMPROVED)
# =========================
class Localizer:
    def __init__(self, detectors):
        self.detectors = detectors
        self.rx_pos = [(0, 0), (5, 0), (5, 5), (0, 5)]

    def estimate(self, real_mask):
        motions = [d.motion() for d in self.detectors]

        min_len = min(len(m) for m in motions)
        motions = [m[:min_len] for m in motions]

        pos = []

        for t in range(min_len):
            if not real_mask[t]:
                continue

            weights = []
            for m in motions:
                val = m[t]
                w = np.exp(val / 10)   # 🔥 improved weighting
                weights.append(w)

            weights = np.array(weights)
            weights /= np.sum(weights)

            x = sum(weights[i] * self.rx_pos[i][0] for i in range(4))
            y = sum(weights[i] * self.rx_pos[i][1] for i in range(4))

            pos.append((x, y))

        return np.array(pos)

# 6. INTERPOLATION
# =========================
def interpolate_path(pos, factor=5):
    if len(pos) < 2:
        return pos

    new_pos = []

    for i in range(len(pos)-1):
        p1 = pos[i]
        p2 = pos[i+1]

        for j in range(factor):
            t = j / factor
            interp = (1 - t) * p1 + t * p2
            new_pos.append(interp)

    new_pos.append(pos[-1])
    return np.array(new_pos)

# =========================
# 7. SMOOTHING
# =========================
def smooth(pos, alpha=0.4):
    if len(pos) == 0:
        return pos
    out = [pos[0]]
    for i in range(1, len(pos)):
        out.append(alpha * pos[i] + (1 - alpha) * out[-1])
    return np.array(out)

def plot_tracking(pos):

    y = pos[:, 1]

    plt.xlabel("X (m)")
    plt.ylabel("Y (m)")
    plt.grid()
    plt.axis('equal')
    plt.legend()

    plt.show()
