from djitellopy import Tello
import cv2
import cv2.aruco as aruco
import numpy as np
import time

# ==========================
# Inisialisasi Tello
# ========================== 
tello = Tello()
tello.connect()
print(f"✅ Baterai: {tello.get_battery()}%")

tello.streamon()
time.sleep(2)

# ==========================
# Setup ArUco
# ==========================
ARUCO_DICT_TYPE = "DICT_5X5_100"
aruco_dict = aruco.getPredefinedDictionary(getattr(aruco, ARUCO_DICT_TYPE))
aruco_params = aruco.DetectorParameters()

detector = aruco.ArucoDetector(aruco_dict, aruco_params)

# ==========================
# KOREOGRAFI KOMPLEKS - 3 MARKER SAJA
# ==========================
def execute_complex_choreography(marker_id):
    """Koreografi kompleks dengan hanya 3 marker"""
    
    if marker_id == 0:
        print("🛑 EMERGENCY LANDING + RETURN TO HOME")
        # Pattern spiral landing dengan return to home
        tello.move_up(50)  # Naik sedikit dulu
        time.sleep(1)
        
        # Spiral search pattern sebelum landing
        for i in range(3):
            tello.move_forward(80)
            tello.rotate_clockwise(120)
            time.sleep(0.5)
        
        # Return to home estimation
        tello.move_back(150)  # Asumsi kembali ke tengah
        tello.move_left(150)  # Asumsi kembali ke tengah
        time.sleep(1)
        
        tello.land()
        return False
    
    elif marker_id == 1:
        print("🚀 FULL ROOM SCAN + OBSTACLE AVOIDANCE PATTERN")
        
        # Phase 1: Vertical scan
        tello.move_up(80)
        time.sleep(1)
        
        # 3D Cube pattern - cover tinggi ruangan
        movements = [
            (100, 0, 0),    # maju
            (0, 100, 0),    # kanan  
            (-100, 0, 0),   # mundur
            (0, -100, 0),   # kiri
            (0, 0, -40),    # turun
            (100, 100, 0),  # diagonal kanan atas
            (-100, -100, 0) # diagonal kiri bawah
        ]
        
        for fwd, right, up in movements:
            if fwd > 0:
                tello.move_forward(abs(fwd))
            elif fwd < 0:
                tello.move_back(abs(fwd))
                
            if right > 0:
                tello.move_right(abs(right))
            elif right < 0:
                tello.move_left(abs(right))
                
            if up > 0:
                tello.move_up(abs(up))
            elif up < 0:
                tello.move_down(abs(up))
                
            time.sleep(0.8)
        
        # Phase 2: Figure 8 dengan rotasi
        print("🔄 Figure 8 dengan camera scan")
        tello.rotate_clockwise(360)  # Scan 360 derajat
        time.sleep(1)
        
        # Complex figure 8 pattern
        figure8_moves = [
            (120, 0), (60, 60), (0, 120), (-60, 60), 
            (-120, 0), (-60, -60), (0, -120), (60, -60)
        ]
        
        for fwd, right in figure8_moves:
            if fwd != 0:
                if fwd > 0:
                    tello.move_forward(abs(fwd))
                else:
                    tello.move_back(abs(fwd))
            
            if right != 0:
                if right > 0:
                    tello.move_right(abs(right))
                else:
                    tello.move_left(abs(right))
            
            # Rotasi kecil setiap gerakan untuk scan visual
            tello.rotate_clockwise(45)
            time.sleep(0.7)
    
    elif marker_id == 2:
        print("🎯 PRECISION PATTERN + SPIRAL EXPANSION")
        
        # Phase 1: Precision zig-zag
        print("📍 Precision Zig-Zag Pattern")
        for i in range(4):
            tello.move_forward(60)
            tello.rotate_clockwise(45 if i % 2 == 0 else -45)
            tello.move_right(40) if i % 2 == 0 else tello.move_left(40)
            time.sleep(0.8)
        
        # Phase 2: Spiral expansion dengan variasi ketinggian
        print("🌀 Spiral Expansion 3D")
        spiral_radius = [30, 60, 90, 120]  # Radius bertahap
        
        for i, radius in enumerate(spiral_radius):
            # Atur ketinggian berbeda setiap spiral
            if i % 2 == 0:
                tello.move_up(20)
            else:
                tello.move_down(20)
            
            # Gerakan spiral
            tello.move_forward(radius)
            tello.move_right(radius)
            tello.move_back(radius)
            tello.move_left(radius)
            
            # Rotasi untuk coverage visual
            tello.rotate_clockwise(90)
            time.sleep(1)
        
        # Phase 3: Final scan pattern
        print("📡 Final 3D Scan")
        # Diamond pattern 3D
        tello.move_forward(100)
        tello.move_up(50)
        tello.move_right(100)
        tello.move_down(50)
        tello.move_back(100)
        tello.move_up(50)
        tello.move_left(100)
        tello.move_down(50)  # Kembali ke posisi awal
        time.sleep(1)
    
    return True

# ==========================
# Kalibrasi untuk kondisi cahaya indoor
# ==========================
def adjust_brightness_contrast(frame, brightness=0, contrast=1.0):
    """Optimasi brightness dan contrast untuk indoor"""
    if brightness != 0:
        if brightness > 0:
            shadow = brightness
            highlight = 255
        else:
            shadow = 0
            highlight = 255 + brightness
        alpha_b = (highlight - shadow) / 255
        gamma_b = shadow
        frame = cv2.addWeighted(frame, alpha_b, frame, 0, gamma_b)
    
    if contrast != 1.0:
        frame = cv2.addWeighted(frame, contrast, frame, 0, 0)
    
    return frame

# ==========================
# Filtering untuk menghindari false positive
# ==========================
last_detected_time = 0
detection_cooldown = 5.0  # Cooldown lebih lama untuk pattern kompleks
current_marker_id = None

def should_process_marker(marker_id):
    """Hindari deteksi berulang terlalu cepat"""
    global last_detected_time, current_marker_id
    
    current_time = time.time()
    if current_time - last_detected_time < detection_cooldown:
        return False
    
    if marker_id != current_marker_id:
        last_detected_time = current_time
        current_marker_id = marker_id
        return True
    
    return False

# ==========================
# Start Video Stream
# ==========================
print("🎥 Mulai deteksi ArUco marker - KOREOGRAFI KOMPLEKS")
print("=== HANYA 3 MARKER DIBUTUHKAN ===")
print("🛑 ID 0: Emergency Landing + Return to Home")
print("🚀 ID 1: Full Room Scan + Obstacle Avoidance")  
print("🎯 ID 2: Precision Pattern + Spiral Expansion")
print("==============================================")
print("Setiap pattern mencakup multiple gerakan kompleks!")
print("Tekan [q] untuk keluar, [ ] untuk takeoff/land")

frame_read = tello.get_frame_read()

# Variabel kontrol manual
manual_control = False
flying = False

try:
    while True:
        frame = frame_read.frame
        
        if frame is None:
            continue
        
        # Pre-processing
        height, width = frame.shape[:2]
        if width > 640:
            frame = cv2.resize(frame, (640, int(height * 640/width)))
        
        processed_frame = adjust_brightness_contrast(frame, brightness=10, contrast=1.2)
        gray = cv2.cvtColor(processed_frame, cv2.COLOR_BGR2GRAY)
        gray = cv2.GaussianBlur(gray, (3, 3), 0)
        
        corners, ids, rejected = detector.detectMarkers(gray)
        
        display = frame.copy()
        
        if ids is not None and len(ids) > 0:
            aruco.drawDetectedMarkers(display, corners, ids)
            
            for i, marker_id in enumerate(ids):
                pts = corners[i][0]
                center_x = int(np.mean(pts[:, 0]))
                center_y = int(np.mean(pts[:, 1]))
                
                cv2.putText(display, f"ID: {marker_id[0]}", (center_x - 20, center_y - 20),
                            cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
                
                # 🧠 OTOMATISASI KOREOGRAFI KOMPLEKS
                if flying and not manual_control and should_process_marker(marker_id[0]):
                    if marker_id[0] in [0, 1, 2]:  # Hanya proses ID 0,1,2
                        still_flying = execute_complex_choreography(marker_id[0])
                        if not still_flying:
                            flying = False
            
            cv2.putText(display, f"Detected ID: {[id[0] for id in ids]}", (20, 30),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
        else:
            cv2.putText(display, "No ArUco Detected", (20, 30),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 0, 255), 2)
        
        cv2.putText(display, f"Flying: {flying}", (20, 60),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0) if flying else (0, 0, 255), 2)
        cv2.putText(display, f"Mode: {'MANUAL' if manual_control else 'AUTO'}", (20, 90),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.7, (255, 255, 0) if manual_control else (0, 255, 255), 2)
        
        # Instruksi
        cv2.putText(display, "SPACE: Takeoff/Land, Q: Quit", (10, display.shape[0] - 60),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.4, (255, 255, 255), 1)
        cv2.putText(display, "ID0: Emergency Land | ID1: Full Scan | ID2: Precision", (10, display.shape[0] - 40),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.4, (0, 255, 255), 1)
        cv2.putText(display, "HANYA 3 MARKER DIBUTUHKAN", (10, display.shape[0] - 20),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.4, (255, 255, 0), 1)
        
        cv2.imshow("Tello Complex Choreography - 3 Markers", display)

        # Kontrol keyboard
        key = cv2.waitKey(1) & 0xFF
        if key == ord('q'):
            break
        elif key == ord(' '):
            if not flying:
                tello.takeoff()
                flying = True
                print("🚀 Takeoff! Siap untuk koreografi kompleks!")
            else:
                tello.land()
                flying = False
                print("🛬 Landing!")
            time.sleep(2)

except KeyboardInterrupt:
    pass
except Exception as e:
    print(f"❌ Error: {e}")

# Cleanup
print("🛑 Mendarat dan menutup...")
if flying:
    tello.land()
tello.streamoff()
cv2.destroyAllWindows()
